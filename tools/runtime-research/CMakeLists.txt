cmake_minimum_required(VERSION 3.16)
project(note4_runtime_research C ASM)

set(CMAKE_C_STANDARD 11)
include(engine.cmake)
add_executable(runtime_probe ${probe_sources} host_main.c)
target_include_directories(runtime_probe PRIVATE "${RESEARCH_GENERATED_DIR}")
target_link_libraries(runtime_probe PRIVATE research_engine)
target_compile_options(runtime_probe PRIVATE -Os -Wall -Wextra)
if(APPLE)
    target_link_options(runtime_probe PRIVATE -Wl,-dead_strip)
else()
    target_link_options(runtime_probe PRIVATE -Wl,--gc-sections)
endif()
